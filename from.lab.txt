    "/quiet"
    "/norestart"
)

#=======[ Execution ]=======#

    New-Item -ItemType Directory -Path $LogFolder -Force | Out-Null
}

$InstallerPath = Join-Path $PSScriptRoot $SourceApp

if (!(Test-Path $InstallerPath)) {
    Write-Host "Installer not found: $InstallerPath"
    Exit 1
}

$Process = Start-Process -FilePath $InstallerPath -ArgumentList $ArgumentList -Wait -PassThru -NoNewWindow

if ($Process.ExitCode -eq 0) {
    Write-Host "Installed $SourceApp successfully"
    Exit 0
}
else {
    Write-Host "Install failed with exit code $($Process.ExitCode)"
    Exit $Process.ExitCode
}
