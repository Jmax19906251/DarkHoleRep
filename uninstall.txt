#=====[ Script Settings ]=====#
$action     = "Install"
$SourceApp  = "BIPublisherDesktop64_64bit.exe"
$SourcePath = Join-Path $PSScriptRoot $SourceApp
$LogFile    = "C:\Windows\Temp\BIPublisher_Install.log"
$LogPath    = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\BIPublisher_${action}.log"

Start-Transcript -Path $LogPath -Append -Force

try {
    Write-Output "Starting $action for BI Publisher Desktop 64-bit..."

    if (-not (Test-Path $SourcePath)) {
        Write-Output "ERROR: EXE not found at $SourcePath"
        Exit 1
    }

    $ArgumentList = "/s /v`"/qn /norestart /L*v `"$LogFile`"`""
    Write-Output "Launching installer: $SourcePath"
    Write-Output "Arguments: $ArgumentList"

    $Process = Start-Process -FilePath $SourcePath -ArgumentList $ArgumentList -Wait -PassThru -WindowStyle Hidden

    Start-Sleep -Seconds 5

    $RegApp = Get-ItemProperty `
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*" ,
        "HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*" `
        -ErrorAction SilentlyContinue |
        Where-Object {
            $_.DisplayName -like "*BI Publisher*" -or
            $_.DisplayName -like "*Template Builder for Word*"
        } |
        Select-Object -First 1

    $FilePaths = @(
        "C:\Program Files\Oracle\BI Publisher\BI Publisher Desktop\Template Builder for Word.xla",
        "C:\Program Files (x86)\Oracle\BI Publisher\BI Publisher Desktop\Template Builder for Word.xla"
    )

    $FileFound = $false
    foreach ($File in $FilePaths) {
        if (Test-Path $File) {
            $FileFound = $true
            Write-Output "Found installed file: $File"
            break
        }
    }

    if ($RegApp -or $FileFound) {
        Write-Output "Installation completed successfully."
        Exit 0
    }
    else {
        Write-Output "Installation failed. App not detected after install."
        Write-Output "Process exit code: $($Process.ExitCode)"
        Exit 1
    }
}
catch {
    Write-Output "ERROR: $($_.Exception.Message)"
    Exit 1
}
finally {
    Stop-Transcript | Out-Null
}
