#=====[ Script Settings ]=====#
$AppName = "Autodesk_MotionBuilder_2026"
$Action  = "Install"

$ScriptDir     = Split-Path -Parent $MyInvocation.MyCommand.Path
$ImageRoot     = Join-Path $ScriptDir "image"
$InstallerExe  = Join-Path $ImageRoot "Installer.exe"
$CollectionXml = Join-Path $ImageRoot "Collection.xml"

$LogPath = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs"
$LogFile = Join-Path $LogPath "${AppName}_${Action}.log"

#=====[ Pre-Flight Checks ]=====#
if (!(Test-Path $InstallerExe)) {
    Write-Error "INSTALLER NOT FOUND: $InstallerExe"
    exit 1
}

if (!(Test-Path $CollectionXml)) {
    Write-Error "COLLECTION XML NOT FOUND: $CollectionXml"
    exit 1
}

if (!(Test-Path $LogPath)) {
    New-Item -Path $LogPath -ItemType Directory -Force | Out-Null
}

#=====[ Execution ]=====#
try {
    "[$((Get-Date).ToString('yyyy-MM-dd HH:mm:ss'))] Starting $Action for $AppName" | Out-File $LogFile -Encoding utf8

    Write-Output "Starting $Action for $AppName"
    Write-Output "Installer Path: $InstallerExe"
    Write-Output "Collection XML: $CollectionXml"

    $ArgumentList = "--ui deploy --offline_mode -q --no_restart -o `"$CollectionXml`""

    Write-Output "Running command: $InstallerExe $ArgumentList"

    $Process = Start-Process -FilePath $InstallerExe `
        -ArgumentList $ArgumentList `
        -WorkingDirectory $ImageRoot `
        -Wait `
        -PassThru `
        -RedirectStandardOutput $LogFile `
        -RedirectStandardError  "$LogFile.stderr.log"

    $ExitCode = $Process.ExitCode

    "[$((Get-Date).ToString('yyyy-MM-dd HH:mm:ss'))] Exit code: $ExitCode" | Out-File $LogFile -Append -Encoding utf8

    if ($ExitCode -eq 0) {
        Write-Output "Install completed successfully. Exit code: $ExitCode"
        exit 0
    }
    elseif ($ExitCode -eq 3010) {
        Write-Warning "Installer returned 3010 (reboot requested) but reboots are suppressed. Exiting 0 — verify app functionality."
        "[$((Get-Date).ToString('yyyy-MM-dd HH:mm:ss'))] WARNING: Exit code 3010 - reboot suppressed." | Out-File $LogFile -Append -Encoding utf8
        exit 0
    }
    else {
        Write-Error "INSTALLATION FAILED. Exit code: $ExitCode. Check log: $LogFile"
        exit $ExitCode
    }
}
catch {
    Write-Error "PowerShell script error occurred: $($_.Exception.Message)"
    "[$((Get-Date).ToString('yyyy-MM-dd HH:mm:ss'))] ERROR: $($_.Exception.Message)" | Out-File $LogFile -Append -Encoding utf8
    exit 1
}
