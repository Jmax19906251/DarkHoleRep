#=====[ Script Settings ]=====#
$AppName     = "SecureFX"
$AppVersion  = "9.6.4.3695"
$LogPath     = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\SecureFX_Install.log"

$Installer   = Join-Path $PSScriptRoot "sfx-x64-bsafe.9.6.4.3695.exe"
$License     = Join-Path $PSScriptRoot "SecureFX.lic"

$InstallDir  = "C:\Program Files\VanDyke Software\SecureFX"
$LicenseDir  = "C:\ProgramData\VanDyke\Config"
$LicenseDest = "$LicenseDir\SecureFX.lic"

$exitCode = 0

Start-Transcript -Path $LogPath -Append -Force

try {
    Write-Output "========================================"
    Write-Output "  $AppName $AppVersion - Install"
    Write-Output "  $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Output "========================================"

    # Pre-flight: verify installer exists
    if (!(Test-Path $Installer)) {
        Write-Output "ERROR: Installer not found at $Installer"
        $exitCode = 1
        throw "Installer missing"
    }

    # Pre-flight: verify license exists
    if (!(Test-Path $License)) {
        Write-Output "ERROR: License file not found at $License"
        $exitCode = 1
        throw "License missing"
    }

    # Create license directory if needed
    if (!(Test-Path $LicenseDir)) {
        Write-Output "Creating license directory: $LicenseDir"
        New-Item -Path $LicenseDir -ItemType Directory -Force | Out-Null
    }

    # Run installer silently
    Write-Output "Running installer: $Installer"
    $proc = Start-Process -FilePath $Installer -ArgumentList "/S" -Wait -NoNewWindow -PassThru

    if ($proc.ExitCode -ne 0) {
        Write-Output "ERROR: Installer exited with code $($proc.ExitCode)"
        $exitCode = $proc.ExitCode
        throw "Installer failed"
    }

    Write-Output "Installer completed successfully (exit code 0)"

    # Copy license file
    Write-Output "Copying license to: $LicenseDest"
    Copy-Item -Path $License -Destination $LicenseDest -Force

    # Verify installation
    if (Test-Path $InstallDir) {
        Write-Output "SUCCESS: $AppName installed at $InstallDir"
        $exitCode = 0
    }
    else {
        Write-Output "ERROR: Install directory not found after install: $InstallDir"
        $exitCode = 1
        throw "Post-install verification failed"
    }
}
catch {
    Write-Output "ERROR: $_"
    if ($exitCode -eq 0) { $exitCode = 1 }
}
finally {
    Write-Output "----------------------------------------"
    Write-Output "Exit Code: $exitCode"
    Write-Output "----------------------------------------"
    Stop-Transcript
}

Exit $exitCode



#=====[ Script Settings ]=====#
$AppName    = "SecureFX"
$LogPath    = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\SecureFX_Uninstall.log"

$InstallDir  = "C:\Program Files\VanDyke Software\SecureFX"
$LicenseDest = "C:\ProgramData\VanDyke\Config\SecureFX.lic"

$UninstallKeys = @(
    "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
)

$exitCode = 0

Start-Transcript -Path $LogPath -Append -Force

try {
    Write-Output "========================================"
    Write-Output "  $AppName - Uninstall"
    Write-Output "  $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
    Write-Output "========================================"

    # Find uninstall string from registry
    $uninstallString = $null
    foreach ($key in $UninstallKeys) {
        $found = Get-ChildItem -Path $key -ErrorAction SilentlyContinue |
            Get-ItemProperty |
            Where-Object { $_.DisplayName -like "*SecureFX*" } |
            Select-Object -First 1 -ExpandProperty UninstallString

        if ($found) {
            $uninstallString = $found
            Write-Output "Found uninstall string: $uninstallString"
            break
        }
    }

    if ($uninstallString) {
        $uninstallExe = $uninstallString.Trim('"').Split(" ")[0]

        Write-Output "Running uninstaller: $uninstallExe"
        $proc = Start-Process -FilePath $uninstallExe -ArgumentList "/S" -Wait -NoNewWindow -PassThru

        if ($proc.ExitCode -ne 0) {
            Write-Output "WARNING: Uninstaller exited with code $($proc.ExitCode)"
        }
        else {
            Write-Output "Uninstaller completed (exit code 0)"
        }
    }
    else {
        Write-Output "No registry uninstall entry found - may already be removed"
    }

    # Remove license file
    if (Test-Path $LicenseDest) {
        Write-Output "Removing license file: $LicenseDest"
        Remove-Item -Path $LicenseDest -Force
    }

    # Verify removal
    if (!(Test-Path $InstallDir)) {
        Write-Output "SUCCESS: $AppName removed"
        $exitCode = 0
    }
    else {
        Write-Output "WARNING: Install directory still exists, forcing removal..."
        Remove-Item -Path $InstallDir -Recurse -Force -ErrorAction SilentlyContinue

        if (!(Test-Path $InstallDir)) {
            Write-Output "SUCCESS: Directory forcefully removed"
            $exitCode = 0
        }
        else {
            Write-Output "ERROR: Unable to remove install directory"
            $exitCode = 1
        }
    }
}
catch {
    Write-Output "ERROR: $_"
    if ($exitCode -eq 0) { $exitCode = 1 }
}
finally {
    Write-Output "----------------------------------------"
    Write-Output "Exit Code: $exitCode"
    Write-Output "----------------------------------------"
    Stop-Transcript
}

Exit $exitCode
