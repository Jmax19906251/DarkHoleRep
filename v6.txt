#=====[ Script Settings ]=====#
$action  = "Uninstall"
$AppName = "PowerBuilder2025"
$LogPath = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\${AppName}_${action}.log"

$VendorUninstall = Join-Path $PSScriptRoot "SilentUninstall.bat"

Start-Transcript -Path $LogPath -Append -Force

try {

Write-Output "Starting $AppName uninstall"

# Run vendor uninstall first
if (Test-Path $VendorUninstall) {

    Write-Output "Running vendor SilentUninstall.bat"

    Start-Process -FilePath "cmd.exe" `
        -ArgumentList "/c `"$VendorUninstall`"" `
        -Wait -NoNewWindow

} else {

    Write-Output "Vendor uninstall script not found"

}

Start-Sleep -Seconds 10


#=====[ Remove leftover folders ]=====#

$Folders = @(
"C:\Program Files\Appeon",
"C:\Program Files (x86)\Appeon",
"C:\ProgramData\Appeon"
)

foreach ($Folder in $Folders) {

    if (Test-Path $Folder) {

        Write-Output "Removing leftover folder $Folder"

        Remove-Item $Folder -Recurse -Force -ErrorAction SilentlyContinue

    }

}


#=====[ Remove leftover uninstall entries ]=====#

$RegPaths = @(
"HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*",
"HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*"
)

foreach ($Path in $RegPaths) {

    $Apps = Get-ItemProperty $Path -ErrorAction SilentlyContinue | Where-Object {
        $_.DisplayName -like "*PowerBuilder*" -or $_.DisplayName -like "*Appeon*"
    }

    foreach ($App in $Apps) {

        Write-Output "Removing uninstall registry entry $($App.DisplayName)"

        Remove-Item -Path $App.PSPath -Recurse -Force -ErrorAction SilentlyContinue

    }

}


Write-Output "$AppName uninstall completed successfully"

Exit 0

}
catch {

Write-Output "ERROR: $($_.Exception.Message)"

Exit 1

}
finally {

Stop-Transcript

}

try {
    Write-Output "Starting $AppName $action"

    Write-Output "Installing MSI..."
    $MsiArgs = "/i `"$MsiFile`" /qn /norestart /L*v `"C:\Windows\Temp\ICUserApps_MSI_Install.log`""

    Write-Output "MSI exit code: $($MsiProcess.ExitCode)"

    if ($MsiProcess.ExitCode -ne 0 -and $MsiProcess.ExitCode -ne 3010) {
        throw "MSI install failed with exit code $($MsiProcess.ExitCode)"
    }

    Write-Output "Installing MSP patch..."
    $MspArgs = "/p `"$MspFile`" /qn /norestart /L*v `"C:\Windows\Temp\ICUserApps_MSP_Install.log`""
    $MspProcess = Start-Process -FilePath "msiexec.exe" -ArgumentList $MspArgs -Wait -PassThru

    Write-Output "MSP exit code: $($MspProcess.ExitCode)"

    if ($MspProcess.ExitCode -ne 0 -and $MspProcess.ExitCode -ne 3010) {
        throw "MSP patch failed with exit code $($MspProcess.ExitCode)"
    }

    Write-Output "$AppName install completed successfully"
    Stop-Transcript
    Exit 0
}
catch {
    Write-Output "ERROR: $($_.Exception.Message)"
    Stop-Transcript
    Exit 1
}

#=====[ Script Settings ]=====#
$action  = "Uninstall"
$AppName = "ICUserApps"
$LogPath = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\${AppName}_${action}.log"

#=====[ Start Logging ]=====#
Start-Transcript -Path $LogPath -Append -Force

try {
    Write-Output "Starting $AppName $action"

    $App = Get-ItemProperty `
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*", `
        "HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*" `
        -ErrorAction SilentlyContinue |
        Where-Object {
            $_.DisplayName -like "*ICUserApps*"
        } |
        Select-Object -First 1

    if ($null -eq $App) {
        Write-Output "$AppName not found. Nothing to uninstall."
        Stop-Transcript
        Exit 0
    }

    Write-Output "Found installed app: $($App.DisplayName)"
    Write-Output "UninstallString: $($App.UninstallString)"

    $Guid = [regex]::Match($App.UninstallString, '\{[A-F0-9\-]+\}', 'IgnoreCase').Value

    if (-not $Guid) {
        throw "Could not find MSI Product Code GUID from uninstall string."
    }

    Write-Output "Uninstalling MSI Product Code: $Guid"

    $UninstallArgs = "/x $Guid /qn /norestart /L*v `"C:\Windows\Temp\ICUserApps_Uninstall.log`""
    $Process = Start-Process -FilePath "msiexec.exe" -ArgumentList $UninstallArgs -Wait -PassThru

    Write-Output "Uninstall exit code: $($Process.ExitCode)"

    if ($Process.ExitCode -ne 0 -and $Process.ExitCode -ne 3010) {
        throw "Uninstall failed with exit code $($Process.ExitCode)"
    }

    Write-Output "$AppName uninstall completed successfully"
    Stop-Transcript
    Exit 0
}
catch {
    Write-Output "ERROR: $($_.Exception.Message)"
    Stop-Transcript
    Exit 1
}
