$path = "HKLM:\SOFTWARE\Policies\Adobe\Acrobat Reader\DC\FeatureLockDown"

# Ensure registry path exists
if (!(Test-Path $path)) {
    New-Item -Path $path -Force | Out-Null
}

# Enable whitelist feature
New-ItemProperty -Path $path -Name "bUseWhitelistConfigFile" -PropertyType DWord -Value 1 -Force | Out-Null

# Adobe install path (Reader)
$installPath = "C:\Program Files (x86)\Adobe\Acrobat Reader DC\Reader"

# Create whitelist config file
$configFile = Join-Path $installPath "ProtectedModeWhitelistConfig.txt"

# Basic test rule (allow read access)
$configContent = @"
; Adobe Protected Mode Whitelist
FILES_ALLOW_READONLY = *
"@

# Create file
$configContent | Out-File -FilePath $configFile -Encoding ASCII -Force

Write-Output "Whitelist enabled and config file created"
exit 0

$path = "HKLM:\SOFTWARE\Policies\Adobe\Acrobat Reader\DC\FeatureLockDown"

# Ensure path exists
if (!(Test-Path $path)) {
    New-Item -Path $path -Force | Out-Null
}

# Disable feature (keep key)
New-ItemProperty -Path $path -Name "bUseWhitelistConfigFile" -PropertyType DWord -Value 0 -Force | Out-Null

# Remove config file
$installPath = "C:\Program Files (x86)\Adobe\Acrobat Reader DC\Reader"
$configFile = Join-Path $installPath "ProtectedModeWhitelistConfig.txt"

if (Test-Path $configFile) {
    Remove-Item $configFile -Force
}

Write-Output "Whitelist disabled and config file removed"
exit 0
