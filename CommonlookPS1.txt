Get-ItemProperty "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*","HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*" |
Where-Object {
    $_.DisplayName -like "*BI Publisher*" -or
    $_.DisplayName -like "*Template Builder for Word*"
} |
Select-Object DisplayName, DisplayVersion, UninstallString, PSPath



Test-Path "C:\Program Files\Oracle\BI Publisher\BI Publisher Desktop\Template Builder for Word.xla"




#=====[ Script Settings ]=====#
$action  = "Uninstall"
$LogPath = "C:\ProgramData\Microsoft\IntuneManagementExtension\Logs\BIPublisherDesktop64Bit_${action}.log"
$AppFile = "C:\Program Files\Oracle\BI Publisher\BI Publisher Desktop\Template Builder for Word.xla"

Start-Transcript -Path $LogPath -Append -Force

try {
    Write-Output "Starting uninstall for BI Publisher Desktop..."

    $App = Get-ItemProperty `
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*" ,
        "HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*" `
        -ErrorAction SilentlyContinue |
        Where-Object {
            $_.DisplayName -like "*BI Publisher*" -or
            $_.DisplayName -like "*Template Builder for Word*"
        } |
        Select-Object -First 1

    if (-not $App) {
        Write-Output "App not found in uninstall registry."
    }
    else {
        Write-Output "Found: $($App.DisplayName)"
        Write-Output "UninstallString: $($App.UninstallString)"

        $UninstallString = $App.UninstallString.Trim()

        if ($UninstallString -match "msiexec") {
            if ($UninstallString -match "({[A-Z0-9\-]+})") {
                $ProductCode = $matches[1]
                $Args = "/x $ProductCode /qn /norestart"
                Write-Output "Running: msiexec.exe $Args"
                $Process = Start-Process -FilePath "msiexec.exe" -ArgumentList $Args -Wait -PassThru -WindowStyle Hidden
            }
            else {
                $Args = $UninstallString.Replace("MsiExec.exe","").Replace("msiexec.exe","").Trim()
                if ($Args -notmatch "/x") {
                    $Args = $Args -replace "/I","/x" -replace "/i","/x"
                }
                if ($Args -notmatch "/qn") {
                    $Args = "$Args /qn /norestart"
                }
                Write-Output "Running: msiexec.exe $Args"
                $Process = Start-Process -FilePath "msiexec.exe" -ArgumentList $Args -Wait -PassThru -WindowStyle Hidden
            }
        }
        else {
            Write-Output "Running EXE uninstall"
            $Process = Start-Process -FilePath "cmd.exe" -ArgumentList "/c `"$UninstallString /s`"" -Wait -PassThru -WindowStyle Hidden
        }

        Write-Output "Exit code: $($Process.ExitCode)"
    }

    Start-Sleep -Seconds 5

    $StillInRegistry = Get-ItemProperty `
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*" ,
        "HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*" `
        -ErrorAction SilentlyContinue |
        Where-Object {
            $_.DisplayName -like "*BI Publisher*" -or
            $_.DisplayName -like "*Template Builder for Word*"
        } |
        Select-Object -First 1

    $FileStillExists = Test-Path $AppFile

    if (-not $StillInRegistry -and -not $FileStillExists) {
        Write-Output "Uninstall completed successfully."
        Exit 0
    }
    else {
        Write-Output "App still detected."
        Write-Output "Registry present: $([bool]$StillInRegistry)"
        Write-Output "File present: $FileStillExists"
        Exit 1
    }
}
catch {
    Write-Output "ERROR: $_"
    Exit 1
}
finally {
    Stop-Transcript | Out-Null
}
